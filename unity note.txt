public static Vector3 StringToVector3(string sVector)
        {
            // Remove the parentheses
            if (sVector.StartsWith("(") && sVector.EndsWith(")"))
            {
                sVector = sVector.Substring(1, sVector.Length - 2);
            }

            // split the items
            string[] sArray = sVector.Split(',');

            // store as a Vector3
            Vector3 result = new Vector3(
                float.Parse(sArray[0]),
                float.Parse(sArray[1]),
                float.Parse(sArray[2]));

            return result;
        }

public string GenerateRandomBinaryString(int length)
    {
        // Ensure the length is an even number for full byte representation if needed,
        // or adjust logic if odd length is truly desired for a specific purpose.
        if (length % 2 != 0)
        {
            Debug.LogWarning("Generating an odd-length hex string may result in an incomplete byte representation.");
        }

        string BinaryString = "";
        for (int i = 0; i < length; i++)
        {
            // Generate a random integer between 0 and 15 (inclusive)
            int randomDigit = Random.Range(0, 9); 
            // Convert the integer to its hexadecimal character representation
            BinaryString += randomDigit.ToString(); 
        }
        return BinaryString;
    }
