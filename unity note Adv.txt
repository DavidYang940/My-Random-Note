    }

    private IEnumerator GetContentsCoroutine(string url, Action<string> callback,Dictionary<string,RequestHeader> requestheader, Action callback2)
    {
        Debug.Log("UnityWebRequest.Get: " + url);
        
        using (UnityWebRequest www = UnityWebRequest.Get(url))
        {
            Debug.Log("Current URL API: " + url);
            www.method="GET";
            if (requestheader != null)
            {
                foreach (var singleheader in requestheader)
                {
                    string requestName = singleheader.Key;
                    string value = singleheader.Value.value;
                    www.SetRequestHeader(requestName,value);
                    
                    //Debug.Log("Header Name: " + requestName + ", Header Value: " + value);
                }
            }
            
            yield return www.SendWebRequest();
            
            //Error handling
            if (www.result == UnityWebRequest.Result.ConnectionError || www.result == UnityWebRequest.Result.ProtocolError)
            {
                Debug.Log("Error connecting to the server: " + www.error + ", " + $"{www.responseCode}");
                /*if(www.isHttpError)
                    Debug.Log($"{www.responseCode}");*/
                
                Debug.Log("Error API: " + url);
            }
            else
            {
                Debug.Log(www.downloadHandler.text);
                callback.Invoke(www.downloadHandler.text);
                //_dataReceived += 1;

                if (debugMod)
                {
                    Debug.Log("Data Received: " + _dataReceived);
                }

                /*if (url == URL + "/vr_bahasa_v2/public/api/v1/area")
                {
                    if (debugMod)
                    {
                        Debug.Log("Contents Loaded");
                    }

                    _onContentsDataReceived?.Invoke();
                }*/

                if (_dataReceived == _dataRequired)
                {
                    if (debugMod)
                    {
                        Debug.Log("All data received");
                    }
                    
                    _onAllDataReceived?.Invoke();
                }
                
                callback2?.Invoke();
            }
        }
    }

 public IEnumerator LoadingAssetBundleCorou(string path,List<AssetBundle> assetBundleContainer,Action eventOnStart=null,Action eventOnComplete=null,Action eventOnFailed=null)
    {
        eventOnStart?.Invoke();
        //unload all asset bundle
        //AssetBundle.UnloadAllAssetBundles(false);
        foreach (var VARIABLE in assetBundleContainer)
        {
            yield return VARIABLE.UnloadAsync(true);
        }
        assetBundleContainer.Clear();
        
        //request load asset bundle with async
        var bundleLoadRequest  = AssetBundle.LoadFromFileAsync(path);

        //wait it complete
        yield return bundleLoadRequest;
        
        //get the asset bundle from request
        assetBundleContainer.Add(bundleLoadRequest.assetBundle);
        
        //check theres asset bundle or no
        if (assetBundleContainer.Count<=0||!assetBundleContainer[0])
        {
            Debug.LogError($"Failed to load AssetBundle with directory :\n{Path.Combine(Application.streamingAssetsPath,path)}");
            eventOnFailed?.Invoke();
            yield break;
        }
        
        //called event after function complete
        eventOnComplete?.Invoke();
    }

//Important----
private string GetAssetBundleName(string assetBundlePath)
    {
        var extensionpath = assetBundlePath + ".manifest";
        using (StreamReader sr = new StreamReader(extensionpath))
        {
            string buffer;
            while ((buffer = sr.ReadLine()) != null)
            {
                if (!buffer.Contains("Assets/")) continue;

                buffer = Path.GetFileName(buffer);
                return buffer;
            }
        }

        return "";
    }
}

    public void OnServerReceivedHeldObjectPosition(NetworkConnectionToClient client,
        HeldObjectPositionsPacket heldObjectPacket)
    {
        var currentStudent =
            DisplayAllUser.Instance.OnlineUserList.FirstOrDefault(x => x.connId == client.connectionId);

        if (currentStudent is null)
            return;

        var roomData = currentStudent.roomData;
        if (roomData is null)
            return;

        var students = roomData.studentList.ToList();

        students.Remove(currentStudent);

        var packetForClient = (HeldObjectPositionsPacketForClient)heldObjectPacket;
        foreach (var student in students)
        {
            student.conn.Send(packetForClient);
        }
    }

    {
        _objectPool = new ObjectPool<V>(createFunc: () => Instantiate(_prefabButtonScript, _parentButton),
            actionOnGet: obj =>
            {
                obj.gameObject.SetActive(true);
                obj.transform.SetAsLastSibling();
            },
            actionOnRelease: obj => obj.gameObject.SetActive(false),
            actionOnDestroy: Destroy,
            collectionCheck: false);
    }
    
    private Dictionary<int, V> _listButtonScript=new Dictionary<int, V>();
    
    public void ClearListButton()
    {
        foreach (var VARIABLE in _listButtonScript)
        {
            _objectPool.Release(VARIABLE.Value);
        }
        _listButtonScript.Clear();
    }
    [SerializeField] protected UnityEvent<int> _selectedDataEvent;
    
    private void Awake()
    {
        InitializeObjectPool();
    }
    
    /// <summary>
    /// update the showed button if theres no button will instantiate
    /// </summary>
    public void SetShowedButton()
    {
        foreach (var VARIABLE in _menuManager.GetShowedData)
        {
            if (_listButtonScript.TryGetValue(VARIABLE.ID, out V result))
            {
                result.gameObject.SetActive(true);
            }
            else
            {
                result = InstantiateButton(VARIABLE);
                _listButtonScript.Add(VARIABLE.ID,result);
            }
        }

        var temp = _menuManager.GetAllData().Except(_menuManager.GetShowedData);

        foreach (var VARIABLE in temp)
        {
            _listButtonScript[VARIABLE.ID].gameObject.SetActive(false);
        }
    }

    /// <summary>
    /// instantiate button and set data to button
    /// </summary>
    /// <param name="data"></param>
    public V InstantiateButton(U data)
    {
        V buttonscript = _objectPool.Get();
        buttonscript.SetData(data);
        buttonscript.SetupUI();
        return buttonscript;
    }
}
